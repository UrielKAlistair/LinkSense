% fig_architecture.tex -- the whole path, with the embedding layer drawn out.
% The embedding layer's two-column horizontal variant is in fig_cellencoder_wide.tex.
%
% Geometry lives in the two blocks below: three x positions, then one y per row.
% Every gap further down is written as "below=Nmm of <node>", edge to edge.
\begin{tikzpicture}[arch, font=\scriptsize]
  % ---- the three columns ----
  \def\cmid{35mm}     % centre line
  \def\clft{16mm}     % left branch
  \def\crgt{54mm}     % right branch
  \def\ctag{56mm}     % the identity-tag box, beside the addition

  % ---- one y per row, down to the summaries; below that the rows are relative.
  % A fork leaves the box above, drops \stub, jogs sideways, and only then runs
  % down into its branch. So the arrow a branch is drawn with is equal above and
  % below only when the branch sits midway between that jog and the join line --
  % \yForkA is the mean of \yFrames minus half its box minus \stub, and \yJoinA.
  % Both branches of a fork must also be the same height, or their arrows differ.
  \def\yCapture{28mm}
  \def\yPrep{16mm}
  \def\yFrames{0mm}
  \def\stub{2mm}       % drop before a fork jogs sideways
  \def\yForkA{-12.5mm}
  \def\yJoinA{-19.75mm}
  \def\yAtt{-30.5mm}
  \def\yForkB{-43mm}
  \def\yJoinB{-50.25mm}
  \def\ySummaries{-54.25mm}   % top edge of both summary boxes, not their centres

  % ---- what reaches the embedding layer ----
  \node[data, text width=26mm] (capture) at (\cmid, \yCapture) {Capture frames + CCA};
  \node[exact, text width=26mm] (prep) at (\cmid, \yPrep) {Preprocessing: binning};

  % ---- frames: content and time embedded on separate paths ----
  \node[data, text width=26mm] (frames) at (\cmid, \yFrames) {Frames in cell (0 to 512)};
  \node[framemlp, text width=21mm] (mlp) at (\clft, \yForkA) {Frame MLP};
  \node[circle, draw=black, fill=exactC!15, minimum size=6.5mm, inner sep=0pt,
        line width=0.5pt] (sin) at (\crgt, \yForkA) {};
  \draw[black, line width=0.5pt] ([xshift=-2.4mm]sin.center)
        sin ([xshift=-1.2mm, yshift=1.1mm]sin.center)
        cos ([xshift=0mm]sin.center)
        sin ([xshift=1.2mm, yshift=-1.1mm]sin.center)
        cos ([xshift=2.4mm]sin.center);
  \node[note, left=1mm of sin, text width=14mm, align=right] (sinlbl)
       {positional encoding of time};
  \node[op] (c1) at (\cmid, \yJoinA) {$\|$};
  \node[selfattn, text width=32mm] (att) at (\cmid, \yAtt) {Self-attention over the frames};

  \draw[flow] (capture) -- (prep);
  \draw[flow] (prep) -- node[dimlbl, right] {cells} (frames);
  \draw[flow] (frames.south) -- ++(0,-\stub) -| (mlp.north);
  \draw[flow] (frames.south) -- ++(0,-\stub) -| (sin.north);
  \draw[flow] (mlp.south) |- (c1.west);
  \draw[flow] (sin.south) |- (c1.east);
  \draw[flow] (c1) -- node[dimlbl, right] {64} (att);
  % centred in the gap each one spans, not along the elbow's total length
\node[dimlbl, above] at ($(c1.center)+(-0.9cm,0)$) {48};
\node[dimlbl, above] at ($(c1.center)+( 0.9cm,0)$) {16};

  % ---- pooling: two reductions, rejoined ----
  \node[meanbox, text width=21mm] (mean) at (\clft, \yForkB) {Mean};
  \node[poolquery, text width=21mm] (query) at (\crgt, \yForkB) {Learned query};
  \node[op] (c2) at (\cmid, \yJoinB) {$\|$};

  \draw[flow] (att.south) -- ++(0,-\stub) -| (mean.north);
  \draw[flow] (att.south) -- ++(0,-\stub) -| (query.north);
  \draw[flow] (mean.south) |- (c2.west);
  \draw[flow] (query.south) |- (c2.east);

  % ---- the two exact summaries: tops aligned, heights left to the text ----
  \node[aggbox, text width=21mm, anchor=north] (agg) at (\clft, \ySummaries)
       {16 aggregates over every frame};
  \node[data, text width=21mm, anchor=north] (cca) at (\crgt, \ySummaries)
       {Carrier sense 109 samples (channel cells only)};
  \node[ccaconv, text width=21mm, below=4.5mm of cca] (conv) {Strided convolution};

  \coordinate (mid) at (\cmid, 0);
  \coordinate (tagx) at (\ctag, 0);
  \coordinate (yjoinc) at ($(conv.south)+(0,-4.75mm)$);
  \node[op] (c3) at (mid |- yjoinc) {$\|$};
  \node[mixmlp, text width=21mm, below=3.75mm of c3] (out) {MLP};
  \node[vector, text width=21mm, below=5mm of out] (cell) {Cell vector};
  \node[op, below=5mm of cell] (c4) {$+$};
  \node[tagbox, text width=23mm] (tags) at (tagx |- c4)
       {Cell tags:\\is\_channel, dwell,\\channel, AP};

  \draw[flow] (c2) -- node[dimlbl, right, pos=0.12] {128} (c3);
  \draw[flow] (agg.south) |- (c3.west);
  \draw[flow] (cca) -- (conv);
  \draw[flow] (conv.south) |- (c3.east);
  \draw[flow] (c3) -- (out);
  \draw[flow] (out) -- node[dimlbl, right] {64} (cell);
  \draw[flow] (cell) -- (c4);
  \draw[flow] (tags) -- (c4);

  % ---- everything above is one reusable unit ----
  \begin{scope}[on background layer]
    \node[draw=black!55, dashed, rounded corners=3pt, inner sep=3mm,
          fit=(frames)(mlp)(sin)(sinlbl)(att)(mean)(query)(agg)(cca)(conv)(out)
              (cell)(c4)(tags)] (unit) {};
  \end{scope}
  \node[note, anchor=north west, font=\scriptsize\itshape]
       at ([shift={(1mm,-1mm)}]unit.north west) {Embedding};

  % ---- what the tokens go on to ----
  % the dashed fork: \optdx across, \optdy up and down, so each line leaves the
  % backbone at atan(\optdy/\optdx) -- 38 degrees as drawn.
  \def\optdx{10mm}
  \def\optdy{6.5mm}
  % The second lane, beside the embedding rather than below it: stacked in one
  % column the figure stands 821pt tall against a 672pt page.
  % The backbone sits above and to the left of what reads it. Its grid steps
  % across in the clear band beneath it and then descends, so that the readout
  % below stands at the centre of the right half of the page, with the grid and
  % the question rows reaching cross-attention equally far either side.
  % The readout stands \cpad further right than the picture alone would allow.
  % The float is centred in what it measures, so reserving that same width off
  % the left edge grows the box evenly and leaves the embedding's place on the
  % page untouched; the text block bounds the pair, at \cpad of about 8mm.
  \def\cpad{10.5mm}     % reserved left of the picture, equal to the shift right
  \def\clane{133mm}    % centre line of the readout, cross-attention and below
  \def\cback{98mm}     % the backbone
  \def\cgrid{124mm}    % the grid's descent, once it has stepped across
  \def\cdesc{142mm}    % the question rows, built beside that descent
  \def\cqtag{157mm}    % the question row's tag box, beside its addition
  \coordinate (lane) at (\cback, \yCapture);
  \coordinate (lanex) at (\clane, 0);
  \coordinate (gridx) at (\cgrid, 0);
  \coordinate (descx) at (\cdesc, 0);
  \coordinate (qtagx) at (\cqtag, 0);
  \node[back, text width=18mm, minimum height=11mm, anchor=north] (backbone)
       at (lane) {Backbone};
  \node[back, font=\scriptsize, text width=24mm, anchor=west]
       (opt1) at ($(backbone.east)+(\optdx,\optdy)$) {Pretrained LLM + LoRA};
  \node[back, font=\scriptsize, text width=24mm, anchor=west]
       (opt2) at ($(backbone.east)+(\optdx,-\optdy)$) {Native transformer encoder};
  % The tokens leave the dashed box at the bottom and carry their label there,
  % then cross to the second lane above both: the corridor between the two
  % dashed boxes is the only place a vertical can run without crossing a node.
  \def\cjog{78mm}     % the crossing, between the embedding and the readout
  \def\yTop{36mm}     % clear of the tallest box in either lane
  \coordinate (leave) at ($(c4.south |- unit.south)+(0,-9mm)$);
  \coordinate (jogx) at (\cjog, 0);
  \coordinate (jog) at (\cjog, \yTop);
  \draw[flow] (c4.south) -- (leave) -- (jogx |- leave) -- (jog)
        -- (backbone.north |- jog) -- (backbone.north);
  \node[dimlbl, anchor=west]
       at ($(c4.south |- unit.south)+(1.5mm,-4.5mm)$) {Embedded cell tokens};
  \draw[dashed, backC] (backbone.east) -- (opt1.west);
  \draw[dashed, backC] (backbone.east) -- (opt2.west);

  % ---- the readout: a question row per AP, reading the grid it never changes ----
  % The grid leaves the backbone to the left and waits there. The rows are built
  % down the centre from the descriptor, and the two meet at cross-attention.
  \node[data, text width=24mm, anchor=north]
       (desc) at ($(backbone.south -| descx)+(0,-15mm)$) {AP descriptor};
  \node[mixmlp, text width=24mm, below=4.5mm of desc] (descmlp) {MLP};
  \node[op, below=5mm of descmlp] (c5) {$+$};
  \node[tagbox, text width=20mm] (qtags) at (qtagx |- c5)
       {Learned query, \\ channel tag,\\ AP tag (joint only)};
  \node[vector, text width=26mm, below=5mm of c5] (qrow) {Question rows, one per AP};

  \node[selfattn, optional, text width=24mm, below=6.5mm of qrow] (dself)
       {Self-attention among rows (joint model only)};
  \node[selfattn, text width=31mm, anchor=north]
       (dcross) at ($(dself.south -| lanex)+(0,-4.5mm)$) {Cross-attention};
  \node[mixmlp, text width=31mm, below=4.5mm of dcross] (dff) {Feed-forward};
  \node[mixmlp, text width=26mm, below=7mm of dff] (head) {Norm and MLP head};
  \node[data, text width=26mm, below=4.5mm of head] (mu) {$(\mu, \log\sigma^2)$ per AP};

  \draw[flow] (desc) -- (descmlp);
  \draw[flow] (descmlp) -- node[dimlbl, right] {64} (c5);
  \draw[flow] (qtags) -- (c5);
  \draw[flow] (c5) -- (qrow);
  \draw[flow] (qrow) -- (dself);
  \draw[flow] (dself.south) -- node[dimlbl, right] {Q}
        (dself.south |- dcross.north);
  \draw[flow] (dcross) -- (dff);
  \draw[flow] (dff) -- (head);
  \draw[flow] (head) -- (mu);
  % The grid is the whole of what the backbone returns; cross-attention derives
  % both its keys and its values from it.
  \draw[flow] (backbone.south) -- (backbone.south |- dself)
        -- (gridx |- dself) -- (gridx |- dcross.north);
  \node[dimlbl, anchor=west, align=center]
       at ($(backbone.south)+(1.5mm,-8.75mm)$) {used for\\K, V};
  % Against the rows' right edge, which stands further out than the centred
  % blocks the brace also spans.
  \coordinate (brcx) at ($(dself.east)+(2mm,0)$);
  \draw[brc, readC] (brcx |- dself.north) -- (brcx |- dff.south)
        node[note, midway, right=2.5mm] {$\times 2$};

  \begin{scope}[on background layer]
    \node[draw=black!55, dashed, rounded corners=3pt, inner sep=3mm,
          fit=(desc)(descmlp)(c5)(qtags)(qrow)(dself)(dcross)(dff)(head)] (rdunit) {};
  \end{scope}
  \node[note, anchor=north west, font=\scriptsize\itshape]
       at ([shift={(1mm,-1mm)}]rdunit.north west) {Readout};

  % Under both lanes rather than under either one: the bounding box is only
  % settled once everything above has been drawn, so this comes last.
  \path ($(current bounding box.west)+(-\cpad,0)$);
  \node[note, below=3mm of current bounding box.south]
       {$\|$ concatenate \qquad $+$ add};
\end{tikzpicture}
